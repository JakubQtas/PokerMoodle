    function czyStritmaly(tab) {
        let t = [...tab].sort((a, b) => a - b);

        return (
            (t[0] == 1 && t[1] == 2 && t[2] == 3 && t[3] == 4 && t[4] == 5)
        );
    }

    function czyStritduzy(tab) {
        let t = [...tab].sort((a, b) => a - b);

        return (
            (t[0] == 2 && t[1] == 3 && t[2] == 4 && t[3] == 5 && t[4] == 6)
        );
    }

    function wynik(wartosci, tab) {

        if (czyStritmaly(tab)) return 4;
        if (czyStritduzy(tab)) return 5;

        switch (wartosci.join(",")) {
            case "5":
                return 8; // poker
            case "4,1":
                return 7; // kareta
            case "3,2":
                return 6; // full
            case "3,1,1":

                return 2; // dwie pary
            case "2,1,1,1":
                return 1; // para
            default:
                return 0; // nic
        }
    }

    function porownajKosci(tab1, tab2) {

        function sortuj(tab) {
            let licznik = {};

            for (let x of tab) {
                licznik[x] = (licznik[x] || 0) + 1;
            }

            return [...tab].sort((a, b) => {
                if (licznik[b] != licznik[a])
                    return licznik[b] - licznik[a];

                return b - a;
            });
        }

        let a = sortuj(tab1);
        let b = sortuj(tab2);

        for (let i = 0; i < 5; i++) {
            if (a[i] > b[i]) return 1;
            if (a[i] < b[i]) return -1;
        }
        
        return 0;
    }
